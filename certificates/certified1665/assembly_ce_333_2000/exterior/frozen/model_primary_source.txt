def ck(s,b):

def a(q):

def rat(x):

def ep(x):return rat(x.lower()),rat(x.upper())
def floor_trunc(x,n):return (F(x)*10**n).__floor__()
def ceil_trunc(x,n):return -floor_trunc(-F(x),n)
def dig(i,n):return ('-' if i<0 else '')+str(abs(i)//10**n)+'.'+str(abs(i)%10**n).zfill(n)
def show(k,x):

def inv(t,k,s):

  for n in range(2,k//d+1):la[d*n]-=la[d]*bt[n]

 for m in range(2,k+1):
  res=sum((la[d]*bt[m//d] for d in range(1,m+1) if m%d==0),arb(0))

 ck('all exact inverse-head identities enclosed',True)

def cfun(x):return ((1+x*x).sqrt()/(4*arb.pi())).log()/4-1/(2*(1+x*x))

def bin_boundaries():

def spatial_block(lo,hi,z):

 else:
